+  const now = new Date();
+  const yyyy = String(now.getFullYear()).padStart(4, "0");

+  const mm = String(now.getMonth() + 1).padStart(2, "0");
+  const seq = String(1).padStart(padding, "0");

+    .replaceAll("{yyyy}", yyyy)
+    .replaceAll("{yy}", yy)
+    .replaceAll("{mm}", mm)
+    .replaceAll("{seq}", seq);
